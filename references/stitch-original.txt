  function closeNavigation(targetSection) {
    const navScreen = document.getElementById('navigation-screen');
    if (navScreen) {
      navScreen.classList.add('closing');
    }
    setTimeout(() => {
      if (window.parent && window.parent !== window) {
        window.parent.postMessage({ type: 'close-navigation', target: targetSection || null }, '*');
      }
      if (document.referrer && document.referrer.includes(window.location.host)) {
        if (targetSection) {
          window.location.href = document.referrer.split('#')[0] + '#' + targetSection;
        } else {
          window.history.back();
        }
      } else if (targetSection) {
        window.location.hash = targetSection;
      }
    }, 240);
  }

<section class="py-20 md:py-28 px-6 border-b border-[#373335] relative" data-purpose="about-editorial" id="about">

    </div>
<div class="max-w-5xl">
<h2 class="font-stencil text-2xl sm:text-3xl md:text-4xl text-[#FEFEFE] leading-snug tracking-tight">
        WADAH PENELITIAN, PENGEMBANGAN DAN PRODUKSI TEKNOLOGI WAHANA TERBANG TANPA AWAK.
      </h2>
<p class="mt-6 text-base md:text-lg text-[#CCC5B2] font-normal leading-relaxed max-w-4xl font-grotesk">
        Semarang State University Flying Object Research (SUVIFOR) merupakan tim riset UAV di bawah naungan UKM Rekayasa Ilmu Pengetahuan dan Teknologi (RIPTEK) Universitas Negeri Semarang. SUVIFOR menjadi wadah mahasiswa yang memiliki minat dan dedikasi dalam bidang UAV dan teknologi kedirgantaraan melalui kegiatan penelitian, pengembangan, edukasi, dan pelatihan. Fokus riset meliputi desain aerodinamika, sistem navigasi otomatis, serta penerapan kecerdasan buatan untuk menghasilkan UAV yang inovatif, efisien, dan aplikatif.
      </p>
</div>
<div class="mt-14 pt-10 border-t border-[#373335] grid grid-cols-1 md:grid-cols-3 gap-10">
<div class="border-l-2 border-[#FACA42] pl-5">
<span class="text-[#FACA42] block font-bold font-mono text-xs tracking-wider uppercase mb-2">01 RISET &amp; PRODUKSI</span>
<p class="text-[#CCC5B2] text-sm leading-relaxed font-grotesk">
          Menjadi pusat penelitian wahana terbang tanpa awak di UNNES serta melakukan proses rekayasa, perancangan, dan produksi sesuai standar industri.
        </p>
</div>
<div class="border-l-2 border-[#FACA42] pl-5">
<span class="text-[#FACA42] block font-bold font-mono text-xs tracking-wider uppercase mb-2">02 EDUKASI &amp; KREASI</span>
<p class="text-[#CCC5B2] text-sm leading-relaxed font-grotesk">
          Menumbuhkan daya tarik mahasiswa terhadap teknologi aviasi melalui pelatihan teknis, simulasi penerbangan, dan pengujian prototipe.
        </p>
</div>
<div class="border-l-2 border-[#FACA42] pl-5">
<span class="text-[#FACA42] block font-bold font-mono text-xs tracking-wider uppercase mb-2">03 KOMPETISI &amp; PRESTASI</span>
<p class="text-[#CCC5B2] text-sm leading-relaxed font-grotesk">
          Menjadi salah satu pemain utama dalam berbagai kegiatan dan kompetisi kedirgantaraan nasional (KRTI) maupun internasional (TEKNOFEST).

<script>
  const openMenuBtn = document.getElementById('open-menu-btn');
  const closeMenuBtn = document.getElementById('close-menu-btn');
  const navOverlay = document.getElementById('nav-overlay');
  const navLinks = document.querySelectorAll('.nav-link');

  function openMenu() {
    navOverlay.classList.add('menu-open');
    navOverlay.setAttribute('aria-hidden', 'false');
    document.body.classList.add('overflow-hidden');
  }

  function closeMenu() {
    navOverlay.classList.remove('menu-open');
    navOverlay.setAttribute('aria-hidden', 'true');
    document.body.classList.remove('overflow-hidden');
  }

  openMenuBtn.addEventListener('click', openMenu);
  closeMenuBtn.addEventListener('click', closeMenu);

  navLinks.forEach(link => {
    link.addEventListener('click', (e) => {
      const targetId = link.getAttribute('href');
      closeMenu();
      if (targetId && targetId.startsWith('#')) {
        const targetElement = document.querySelector(targetId);
        if (targetElement) {
          setTimeout(() => {
            targetElement.scrollIntoView({ behavior: 'smooth' });
          }, 320);
        }
      }
    });
  });

  // Support ESC key to dismiss menu
  document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape' && navOverlay.classList.contains('menu-open')) {
      closeMenu();
    }
  });
</script>
